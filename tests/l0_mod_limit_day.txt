# speed level 2. Eight visitors reach 20 mod objects arrived today (themselves included) within two rounds, and the next call stops the
# game with a message that names the function and the line of the script (a world error stops the run).
# @env OBL_PACK=l0
# @env OBL_MODS=l0_limits
# @timeout 120

# @expect-log FatalError: .*Function CreateObject: the limit of 20 mod objects created a day
seed 1

await gameplay
command set-game-speed level=2
step 1
spawn-guests 8
step 2000
